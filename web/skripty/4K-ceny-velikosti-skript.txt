<script>
/* 4K – cena pod každou velikostí na detailu zboží (jako ve vratkovém portálu).
   Ceny bere z JSON-LD stránky (hasVariant → offers.price), dlaždici najde podle konce názvu varianty
   nebo podle konce názvu obrázku (…-42-5.jpg). Patří na konec Zápatí. */
(function () {
  function vars() {
    var s = document.querySelectorAll('script[type="application/ld+json"]');
    for (var i = 0; i < s.length; i++) {
      try { var j = JSON.parse(s[i].textContent); if (j && j.hasVariant && j.hasVariant.length) return j.hasVariant; } catch (e) {}
    }
    return [];
  }
  var slug = function (s) { return String(s).toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, ''); };
  var nf = function (n) { return Math.round(n).toLocaleString('cs-CZ').replace(/\s/g, '\u00a0') + '\u00a0Kč'; };
  function price(v) { var o = v.offers; if (o && o.length) o = o[0]; var p = o ? parseFloat(o.price) : NaN; return p > 0 ? p : null; }
  function run() {
    var tiles = document.querySelectorAll('#PDVariantSelector .buttons .value[data-value]');
    if (tiles.length < 2) return;
    var map = {}, vs = vars();
    for (var i = 0; i < vs.length; i++) {
      var p = price(vs[i]); if (p === null) continue;
      var last = String(vs[i].name || '').trim().split(/\s+/).pop().replace(/^vel\.?/i, '').toLowerCase();
      var file = slug(String(vs[i].image || '').split('?')[0].split('/').pop().replace(/\.[a-z0-9]+$/i, ''));
      var hit = null, len = 0;
      for (var k = 0; k < tiles.length; k++) {
        var val = tiles[k].getAttribute('data-value'), sv = slug(val);
        if (val.toLowerCase() === last) { hit = val; break; }
        if (sv && sv.length > len && file.slice(-sv.length - 1) === '-' + sv) { hit = val; len = sv.length; }
      }
      if (hit !== null && map[hit] === undefined) map[hit] = p;
    }
    // jen když se ceny velikostí liší – při stejné ceně u všech nic nepřidávat
    var seen = {}, n = 0;
    for (var key in map) if (!seen[Math.round(map[key])]) { seen[Math.round(map[key])] = 1; n++; }
    if (n < 2) map = {};
    for (var t = 0; t < tiles.length; t++) {
      var v = tiles[t].getAttribute('data-value'), el = tiles[t].querySelector('.k4-cena');
      if (map[v] === undefined) { if (el) el.remove(); continue; }
      if (!el) { el = document.createElement('span'); el.className = 'k4-cena'; tiles[t].appendChild(el); }
      if (el.textContent !== nf(map[v])) el.textContent = nf(map[v]);
    }
  }
  var css = document.createElement('style');
  css.textContent = '#PDVariantSelector .buttons .value{height:auto!important;min-height:69px}#PDVariantSelector .buttons .value .k4-cena{display:block;font-size:13px;line-height:1.3;color:#7a7a76;margin-top:2px}#PDVariantSelector .buttons .value.selected .k4-cena{color:#d6d6d2}';
  document.head.appendChild(css);
  run();
  var t = null;
  new MutationObserver(function () { clearTimeout(t); t = setTimeout(run, 150); })
    .observe(document.body, { childList: true, subtree: true });
})();
</script>
